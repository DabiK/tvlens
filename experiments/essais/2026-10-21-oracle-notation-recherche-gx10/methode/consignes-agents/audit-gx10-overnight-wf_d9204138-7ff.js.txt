// The Mac allows at most four subagents at once: hand slots over directly so the cap is never exceeded.
const MAX = 4
let active = 0
const waiting = []
async function limited(fn) {
  if (active >= MAX) { await new Promise(r => waiting.push(r)) } else { active++ }
  try { return await fn() } finally { const next = waiting.shift(); if (next) next(); else active-- }
}

const AUDIT_SCHEMA = {
  type: 'object',
  properties: {
    unit: { type: 'string' },
    claims: { type: 'array', items: { type: 'object', properties: {
      claim: { type: 'string' }, source: { type: 'string' }, recomputed: { type: 'string' },
      status: { type: 'string', enum: ['confirmed', 'discrepancy', 'unverifiable'] }, evidence: { type: 'string' } },
      required: ['claim', 'status', 'evidence'] } },
    seals: { type: 'array', items: { type: 'object', properties: {
      seal: { type: 'string' }, filesChecked: { type: 'number' }, ok: { type: 'boolean' }, detail: { type: 'string' } },
      required: ['seal', 'ok', 'detail'] } },
    tests: { type: 'array', items: { type: 'object', properties: {
      command: { type: 'string' }, result: { type: 'string' } }, required: ['command', 'result'] } },
    issues: { type: 'array', items: { type: 'object', properties: {
      id: { type: 'string' }, severity: { type: 'string', enum: ['high', 'medium', 'low'] },
      kind: { type: 'string' }, location: { type: 'string' }, description: { type: 'string' }, evidence: { type: 'string' } },
      required: ['id', 'severity', 'kind', 'location', 'description', 'evidence'] } },
    priorityQuestions: { type: 'array', items: { type: 'object', properties: {
      id: { type: 'string' }, currentExpected: { type: 'string' }, localEvidence: { type: 'string' },
      candidateSupported: { type: 'string' }, needsMedia: { type: 'string' } }, required: ['id', 'localEvidence'] } },
    summary: { type: 'string' },
  },
  required: ['unit', 'claims', 'issues', 'summary'],
}

const VERDICT_SCHEMA = {
  type: 'object',
  properties: {
    refuted: { type: 'boolean' },
    correctedSeverity: { type: 'string', enum: ['high', 'medium', 'low', 'none'] },
    reasoning: { type: 'string' },
    evidence: { type: 'string' },
  },
  required: ['refuted', 'correctedSeverity', 'reasoning', 'evidence'],
}

function auditPrompt(u) {
  const where = u.dirs.length ? u.dirs.map(d => `${ESS}/${d}`).join('\n') : '(transversal, voir la consigne)'
  return `${RULES}
Ton unité : « ${u.key} ». Dossier de travail : ${SCR}/${u.key}/
Dossiers à auditer :
${where}

Ce qu'il faut vérifier en priorité :
${u.focus}

Rends : chaque affirmation chiffrée vérifiée (claims), les sceaux contrôlés (seals), les tests lancés avec leur sortie résumée (tests),
et les problèmes (issues) avec un id court unique préfixé par « ${u.key}- ». Sévérité high = un chiffre ou une conclusion du rapport
est faux ou non reproductible, ou un secret/une adresse privée est suivi par Git ; medium = écart matériel, trace manquante, aveugle
fragile ; low = forme, arrondi, lien cassé.`
}

function verifyPrompt(u, issue) {
  return `${RULES}
Tu es un vérificateur SCEPTIQUE. Un auditeur de l'unité « ${u.key} » affirme le problème suivant. Ton travail : essayer de le RÉFUTER
en retournant aux fichiers bruts toi-même (ne fais pas confiance à l'auditeur). Dossier de travail : ${SCR}/verify-${issue.id}/

Problème ${issue.id} (sévérité annoncée ${issue.severity}, type ${issue.kind})
Emplacement : ${issue.location}
Description : ${issue.description}
Preuve donnée : ${issue.evidence}

Dis refuted=true si le problème n'existe pas ou repose sur une mauvaise lecture (autre ensemble, autre passage, arrondi légitime,
fichier explicitement hors Git par conception documentée…). Sinon refuted=false et donne la sévérité corrigée. En cas de doute réel,
refuted=false avec la sévérité la plus basse défendable. Donne ta preuve (chemins, valeurs).`
}

phase('Audit')
const results = await pipeline(
  UNITS,
  u => limited(() => agent(auditPrompt(u), { label: `audit:${u.key}`, phase: 'Audit', schema: AUDIT_SCHEMA })),
  (audit, u) => {
    if (!audit) return null
    const toCheck = (audit.issues || []).filter(i => i.severity !== 'low')
    const lows = (audit.issues || []).filter(i => i.severity === 'low')
    if (lows.length) log(`${u.key}: ${lows.length} constat(s) mineur(s) non contre-vérifié(s)`)
    return parallel(toCheck.map(i => () =>
      limited(() => agent(verifyPrompt(u, i), { label: `verify:${i.id}`, phase: 'Verify', schema: VERDICT_SCHEMA }))
        .then(v => ({ ...i, verdict: v }))))
      .then(verified => ({ ...audit, verifiedIssues: verified.filter(Boolean), lowIssues: lows }))
  },
)
return results.filter(Boolean)
