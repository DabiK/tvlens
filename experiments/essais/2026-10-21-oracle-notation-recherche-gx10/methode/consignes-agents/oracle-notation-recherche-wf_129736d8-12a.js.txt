const A = args
const MAX = 4
let active = 0
const waiting = []
async function limited(fn) {
  if (active >= MAX) { await new Promise(r => waiting.push(r)) } else { active++ }
  try { return await fn() } finally { const next = waiting.shift(); if (next) next(); else active-- }
}

function reviewerPrompt(v, which) {
  return `${COMMON}
Tu es le relecteur ${which} (indépendant de l'autre relecteur) de l'oracle v2 de la vidéo « ${v.key} » (${v.lang}). ${v.note}
Dossier de travail : ${A.lab}/oracle/travail/${v.key}-${which}/ (crée-le).
Fiche v1 : snapshots/${v.fiche}.* ; questions : ${v.fiche}.json. Il y a exactement cinq questions évaluées : celles du fichier
${A.lab}/notation/a-noter-${v.key}.json (lis UNIQUEMENT les champs questionId, atMs et question de ce fichier ; n'en lis PAS les
réponses — tu ne dois jamais voir une réponse d'un modèle évalué).

INTERDIT : ouvrir des campaign.json, des dossiers rapatriements/, annotation*, correspondance*, cle-*, membres-*, ou toute sortie de
TVLens. L'attendu se construit depuis les médias seulement.

Pour chaque question TELLE QU'ELLE A ÉTÉ POSÉE (texte et instant inchangés) :
1. Retrouve dans la transcription et les images ce qui répond, avec horodatage précis ; regarde les images autour (planches).
2. Liste les faits requis (ceux qu'une bonne réponse doit contenir, disponibles A) et optionnels (B, ou détails), avec modalité et
   disponibilité selon la règle commune. Un fait C ne peut pas être requis.
3. Dis si la question est répondable à la pose (yes/partial/no) ; si non, la politique d'abstention attendue.
4. Liste les inventions typiques à sanctionner (faits plausibles mais faux, anticipations, connaissances externes).
5. Écris une grille 0–3 propre à la question (s3, s2, s1, s0), en tenant compte de la langue de la question.
6. Compare à la fiche v1 : confirmed, corrected (attendu ajusté) ou flawed (question mal posée à cet instant), et explique.
7. Recommandation pour une future version (pose, formulation), sans l'appliquer à la notation.
Sois précis et vérifie tes horodatages ; indique ta confiance.`
}

  return `${COMMON}
Tu es l'arbitre de l'oracle v2 de la vidéo « ${v.key} » (${v.lang}). ${v.note}
Deux relecteurs indépendants ont produit les propositions ci-dessous. Réconcilie-les en revenant aux médias pour chaque point de
désaccord (horodatage, fait requis ou non, disponibilité, grille). Ne vois aucune réponse de modèle évalué (mêmes interdits que les
relecteurs : pas de campaign.json, rapatriements, annotation, correspondance, cle-*, membres-*). Rends l'oracle final des cinq
questions, au même format. Dans notes, liste les désaccords et comment tu les as tranchés. Dossier : ${A.lab}/oracle/travail/${v.key}-arbitre/

Relecteur 1 :
${JSON.stringify(r1)}

Relecteur 2 :
${JSON.stringify(r2)}`
}

function judgePrompt(v, oracle, which, subset) {
  const scope = subset ? `Ne note QUE ces textId : ${JSON.stringify(subset)}.` : 'Note tous les textes du paquet.'
  return `Tu es le juge ${which}, indépendant, de réponses d'un compagnon de visionnage, vidéo « ${v.key} ». Notation à l'aveugle.
Paquet : ${A.lab}/notation/a-noter-${v.key}.json (questions et réponses anonymisées ; les réponses de type « insufficient » sont
des abstentions du système). ${scope}
INTERDIT : ouvrir cle-*, membres-*, correspondance*, campaign.json, rapatriements, ou chercher d'où vient une réponse. Ne lis pas le
dépôt TVLens. Tu peux vérifier un fait douteux dans la transcription ${A.lab}/oracle/transcriptions/${v.key}.transcript.txt ou les
images ${A.media}/frames/${v.key}/ (outil planche : python3 -I ${A.lab}/planche.py --frames … --from s --to s --out f.jpg).

Oracle v2 de la vidéo (faits requis/optionnels, disponibilité à la pose, pièges d'invention, politique d'abstention, grille) :
${JSON.stringify(oracle)}

Pour chaque texte : score 0–3 selon la grille de sa question ; invention=true si la réponse affirme un fait faux ou non soutenu par
le média (une anticipation de ce qui n'a pas encore eu lieu est une invention) ; abstention : none si le texte répond,
justified si abstention alors que la question n'est pas répondable à la pose (selon l'oracle), unjustified sinon (score 0) ;
temporalEvidenceOk : les citations (startMs/endMs) pointent-elles vers des moments qui soutiennent la réponse (na si aucune) ;
factsCovered ; rationale courte et précise. Une réponse dans une autre langue que la question n'est pas fausse pour autant, mais
note-le. Sois constant entre les réponses d'une même question.`
}

// Oracle → judging pipeline per video, research in parallel; all agents share the 4-slot limiter.
const oracleAndScores = pipeline(
  VIDEOS,
  v => parallel([1, 2].map(n => () =>
    limited(() => agent(reviewerPrompt(v, n), { label: `oracle:${v.key}:r${n}`, phase: 'Oracle', schema: ORACLE_SCHEMA })))),
  (reviews, v) => {
    const [r1, r2] = reviews
    if (!r1 || !r2) return null
    return limited(() => agent(adjudicatorPrompt(v, r1, r2), { label: `oracle:${v.key}:arbitre`, phase: 'Oracle', schema: ORACLE_SCHEMA }))
  },
  async (oracle, v) => {
    if (!oracle) return null
    const judges = await parallel([1, 2].map(n => () =>
      limited(() => agent(judgePrompt(v, oracle, n, null), { label: `juge:${v.key}:${n}`, phase: 'Notation', schema: JUDGE_SCHEMA }))))
    const [j1, j2] = judges
    let j3 = null
    if (j1 && j2) {
      const s2 = new Map(j2.scores.map(s => [s.textId, s]))
      const disputed = j1.scores.filter(s => {
        const o = s2.get(s.textId)
        return !o || Math.abs(o.score - s.score) >= 2 || o.invention !== s.invention
      }).map(s => s.textId)
      if (disputed.length) {
        log(`${v.key}: ${disputed.length} désaccord(s) tranché(s) par un troisième juge`)
        j3 = await limited(() => agent(judgePrompt(v, oracle, 3, disputed), { label: `juge:${v.key}:3`, phase: 'Notation', schema: JUDGE_SCHEMA }))
      }
    }
    return { video: v.key, oracle, judges: [j1, j2, j3] }
  },
)

const research = (async () => {
  const reviews = await parallel(AXES.map(axis => () =>
    limited(() => agent(researchPrompt(axis), { label: `recherche:${axis.key}`, phase: 'Recherche', schema: RESEARCH_SCHEMA }))))
  const ok = reviews.filter(Boolean)
  const synthesis = ok.length ? await limited(() => agent(synthesisPrompt(ok), { label: 'recherche:synthese', phase: 'Recherche' })) : null
  return { reviews: ok, synthesis }
})()

const [scored, researched] = await Promise.all([oracleAndScores, research])
return { scored: scored.filter(Boolean), research: researched }
